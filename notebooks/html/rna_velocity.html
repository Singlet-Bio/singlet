RNA Velocity Matrices¶

Load spliced and unspliced count matrices from singlify output for RNA velocity analysis. The pipeline produces separate exon (spliced) and intron (unspliced) count matrices ready for scVelo or velocyto.

Sample: GSM3573650 (human bone marrow, ~75K cells)

import singlet
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import scipy.sparse as sp
import os

plt.rcParams['figure.dpi'] = 120

sample_dir = '/mnt/projects/debruinz_project/singlify_pipeline/quant/scrna/GSE125/GSE125416/GSM3573650'

# Load exon-level and intron-level matrices
exon_adata = singlet.read_1pz(os.path.join(sample_dir, 'exon_counts.1pz'))
intron_adata = singlet.read_1pz(os.path.join(sample_dir, 'intron_counts.1pz'))

print(f'Raw exon features: {exon_adata.n_vars:,} (exon-level)')
print(f'Raw intron features: {intron_adata.n_vars:,} (intron-level)')

# Aggregate to gene level: extract ENSG ID from feature names
# Exon format: ENSG00000000003_TSPAN6_chrX:100627107-100629986
# Intron format: ENSG00000000003_TSPAN6_intron_chrX:100629986-100630758
exon_genes = [n.split('_')[0] for n in exon_adata.var_names]
intron_genes = [n.split('_')[0] for n in intron_adata.var_names]

# Build gene-level aggregation matrices
from collections import OrderedDict
def aggregate_to_genes(adata, gene_ids):
    unique_genes = list(OrderedDict.fromkeys(gene_ids))
    gene_to_idx = {g: i for i, g in enumerate(unique_genes)}
    row_idx = list(range(len(gene_ids)))
    col_idx = [gene_to_idx[g] for g in gene_ids]
    agg_mat = sp.csc_matrix((np.ones(len(gene_ids)), (col_idx, row_idx)),
                           shape=(len(unique_genes), len(gene_ids)))
    gene_mat = agg_mat @ adata.X.T  # genes x cells
    return gene_mat.T, unique_genes  # cells x genes

exon_mat, exon_gene_names = aggregate_to_genes(exon_adata, exon_genes)
intron_mat, intron_gene_names = aggregate_to_genes(intron_adata, intron_genes)

# Intersect on common genes
common = sorted(set(exon_gene_names) & set(intron_gene_names))
exon_idx = [exon_gene_names.index(g) for g in common]
intron_idx = [intron_gene_names.index(g) for g in common]
exon_mat = exon_mat[:, exon_idx]
intron_mat = intron_mat[:, intron_idx]

print(f'\nGene-level aggregation:')
print(f'  Spliced (exon): {exon_mat.shape[0]:,} cells x {exon_mat.shape[1]:,} genes')
print(f'  Unspliced (intron): {intron_mat.shape[0]:,} cells x {intron_mat.shape[1]:,} genes')
print(f'\nTotal spliced UMIs: {exon_mat.sum():,.0f}')
print(f'Total unspliced UMIs: {intron_mat.sum():,.0f}')
print(f'Overall intronic fraction: {intron_mat.sum() / (exon_mat.sum() + intron_mat.sum()) * 100:.1f}%')

# Per-cell velocity readiness
cell_spliced = np.array(exon_mat.sum(axis=1)).flatten()
cell_unspliced = np.array(intron_mat.sum(axis=1)).flatten()
cell_total = cell_spliced + cell_unspliced
velocity_ready = (cell_spliced > 100) & (cell_unspliced > 10)

print(f"Cells with >100 spliced + >10 unspliced UMIs: {velocity_ready.sum():,} / {len(velocity_ready):,}")
print(f"Velocity-ready fraction: {velocity_ready.mean()*100:.1f}%")

fig, axes = plt.subplots(1, 2, figsize=(12, 5))

ax = axes[0]
ax.scatter(cell_spliced, cell_unspliced, alpha=0.02, s=2, c='#3b82f6')
ax.set_xlabel('Spliced UMIs/cell')
ax.set_ylabel('Unspliced UMIs/cell')
ax.set_title(f'Spliced vs Unspliced per Cell\n({velocity_ready.sum():,} velocity-ready)')

print('''
# To use with scVelo:

import scvelo as scv
adata = singlet.load_dir(sample_dir)
adata.layers['spliced'] = exon_mat
adata.layers['unspliced'] = intron_mat

scv.pp.filter_and_normalize(adata, min_shared_counts=20)
scv.pp.moments(adata, n_pcs=30, n_neighbors=30)
scv.tl.velocity(adata)
scv.tl.velocity_graph(adata)
scv.pl.velocity_embedding_stream(adata, basis='umap')
''')
